cmake_minimum_required(VERSION 3.24)
project(StrikeTrainingLab VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(strike-trainer train_bots.cpp)
add_executable(strike-actual-trainer train_actual.cpp)
add_executable(training-tests tests.cpp)
if(MSVC)
  target_compile_options(strike-trainer PRIVATE /W4)
  target_compile_options(training-tests PRIVATE /W4)
else()
  target_compile_options(strike-trainer PRIVATE -Wall -Wextra)
  target_compile_options(training-tests PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME training-mechanics COMMAND training-tests)
