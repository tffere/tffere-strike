cmake_minimum_required(VERSION 3.24)
project(StrikeTrainingLab VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(strike-trainer train_bots.cpp)
add_executable(strike-actual-trainer train_actual.cpp)
add_executable(strike-actual-evaluate evaluate_actual.cpp)
add_executable(strike-real-dual train_real_dual.cpp)
add_executable(strike-diagnose diagnose_real.cpp)
add_executable(strike-campaign train_campaign.cpp)
add_executable(training-tests tests.cpp)
if(MSVC)
  target_compile_options(strike-trainer PRIVATE /W4)
  target_compile_options(training-tests PRIVATE /W4)
else()
  target_compile_options(strike-trainer PRIVATE -Wall -Wextra)
  target_compile_options(training-tests PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME training-mechanics COMMAND training-tests)

add_executable(campaign-tests campaign_tests.cpp)
add_test(NAME campaign-metrics COMMAND campaign-tests)

add_executable(strike-frozen-audit audit_frozen.cpp)
add_executable(strike-state-audit audit_state_aliasing.cpp)
add_executable(strike-timing-audit audit_timing.cpp)
add_executable(strike-symmetric-audit audit_symmetric_duel.cpp)

add_executable(strike-context-trainer train_context.cpp)
add_executable(context-tests context_tests.cpp)
add_test(NAME contextual-policy COMMAND context-tests)
